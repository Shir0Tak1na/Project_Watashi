# Local-only stack for the Project Watashi prototype.
#
# Everything here runs offline. The ONNX weights ship inside
# rapidocr-onnxruntime, so no model download is needed and no cloud call is
# made (requirement R1).
#
# tkinter is used for the overlay and is part of the Python standard library,
# so it is intentionally not listed.
#
# This file is what `pip install -r` installs, and therefore what CI installs.
# selfcheck_deps exists because it was missing two packages that the local venv
# happened to have: a check that passes locally and cannot start in a clean
# environment is the failure mode that file guards against.

rapidocr-onnxruntime>=1.4.4
onnxruntime>=1.17
opencv-python>=4.8
numpy>=1.24
mss>=9.0
pillow>=10.0
pyyaml>=6.0

# The web settings panel. Declared rather than left to the local environment:
# watashi/web.py imports fastapi when the module is loaded, so without it
# selfcheck_web cannot even be imported, and uvicorn is what actually serves the
# panel -- the import is guarded, but the check starts a real server, so the
# guard turns a missing package into a failed assertion instead of a clear error.
fastapi>=0.110
uvicorn>=0.29

# Linux capture. mss declares no runtime dependencies of its own, but its Linux
# backend imports Xlib, so without this a Linux user gets an ImportError the first
# time anything is captured. The CI checks never capture, so this is for people
# running the prototype rather than for the badge.
python-xlib>=0.33; sys_platform == "linux"

# Optional: local LLM backend for M3 (requirement R1). Requires a local GGUF
# model file and is enabled by setting translation.llm_model in config.yaml.
# No cloud API is used.
# llama-cpp-python>=0.2.0

# Local neural machine translation (requirement R1 / M3). CTranslate2 runs the
# model on CPU and tokenizers reads the NLLB tokenizer. Model weights are
# fetched separately by fetch_model.py, not by pip.
ctranslate2>=4.5
sentencepiece>=0.2
tokenizers>=0.20

# Used by fetch_model.py for the one-time model download only; the running
# application never needs the network.
requests>=2.31
